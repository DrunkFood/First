```java
package com.jy.eleaitender.file.engine;

import com.jy.eleaitender.common.dto.response.WordStructureVO;
import lombok.extern.slf4j.Slf4j;
import org.apache.poi.xwpf.usermodel.*;
import org.springframework.stereotype.Component;

import java.io.InputStream;
import java.util.*;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

@Slf4j
@Component
public class WordStructureParser {

    private static final Pattern PLACEHOLDER_PATTERN = Pattern.compile("\\{\\{(\\w+)}}");

    public WordStructureVO parse(InputStream inputStream) {
        WordStructureVO vo = new WordStructureVO();
        List<WordStructureVO.Chapter> chapters = new ArrayList<>();
        Set<String> placeholders = new LinkedHashSet<>();
        Set<String> bookmarks = new LinkedHashSet<>();

        try (XWPFDocument doc = new XWPFDocument(inputStream)) {
            for (IBodyElement element : doc.getBodyElements()) {
                if (element instanceof XWPFParagraph para) {
                    extractFromParagraph(para, chapters, placeholders, bookmarks);
                } else if (element instanceof XWPFTable table) {
                    extractFromTable(table, placeholders, bookmarks);
                }
            }
        } catch (Exception e) {
            log.error("解析Word文档结构失败", e);
            throw new RuntimeException("解析Word文档结构失败: " + e.getMessage(), e);
        }

        vo.setChapters(chapters);
        vo.setPlaceholders(new ArrayList<>(placeholders));
        vo.setBookmarks(new ArrayList<>(bookmarks));
        return vo;
    }

    private void extractFromParagraph(XWPFParagraph para, List<WordStructureVO.Chapter> chapters,
                                       Set<String> placeholders, Set<String> bookmarks) {
        String style = para.getStyle();
        String text = para.getText();

        if (style != null && style.startsWith("Heading")) {
            int level = parseHeadingLevel(style);
            WordStructureVO.Chapter chapter = new WordStructureVO.Chapter();
            chapter.setLevel(level);
            chapter.setTitle(text != null ? text.trim() : "");
            chapter.setBookmarks(new ArrayList<>());
            chapters.add(chapter);
        }

        if (text != null) {
            Matcher matcher = PLACEHOLDER_PATTERN.matcher(text);
            while (matcher.find()) {
                placeholders.add(matcher.group(1));
            }
        }

        for (CTBookmark bm : para.getCTP().getBookmarkStartList()) {
            String bmName = bm.getName();
            if (bmName != null && !bmName.startsWith("_")) {
                bookmarks.add(bmName);
                if (!chapters.isEmpty()) {
                    WordStructureVO.Chapter lastChapter = chapters.get(chapters.size() - 1);
                    lastChapter.getBookmarks().add(bmName);
                }
            }
        }
    }

    private void extractFromTable(XWPFTable table, Set<String> placeholders, Set<String> bookmarks) {
        for (XWPFTableRow row : table.getRows()) {
            for (XWPFTableCell cell : row.getTableCells()) {
                for (XWPFParagraph para : cell.getParagraphs()) {
                    String text = para.getText();
                    if (text != null) {
                        Matcher matcher = PLACEHOLDER_PATTERN.matcher(text);
                        while (matcher.find()) {
                            placeholders.add(matcher.group(1));
                        }
                    }
                    for (CTBookmark bm : para.getCTP().getBookmarkStartList()) {
                        String bmName = bm.getName();
                        if (bmName != null && !bmName.startsWith("_")) {
                            bookmarks.add(bmName);
                        }
                    }
                }
            }
        }
    }

    private int parseHeadingLevel(String style) {
        try {
            return Integer.parseInt(style.replace("Heading", "").trim());
        } catch (NumberFormatException e) {
            return 1;
        }
    }
}
```

注意：`CTBookmark` 需要 `import org.openxmlformats.schemas.wordprocessingml.x2006.main.CTBookmark;`，该类在 poi-tl 的传递依赖 Apache POI OOXML 中。

```java
package com.jy.eleaitender.file.engine;

import com.deepoove.poi.XWPFTemplate;
import com.deepoove.poi.config.Configure;
import com.deepoove.poi.config.ConfigureBuilder;
import com.deepoove.poi.plugin.bookmark.BookmarkRenderPolicy;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.io.ByteArrayOutputStream;
import java.io.InputStream;
import java.util.Map;

@Slf4j
@Component
public class WordTemplateEngine {

    public byte[] render(InputStream templateStream, Map<String, Object> data) {
        log.info("开始Word模板填充，数据字段数: {}", data.size());

        ConfigureBuilder builder = Configure.builder();
        builder.useSpringEL();

        Configure config = builder.build();

        try (InputStream is = templateStream;
             ByteArrayOutputStream out = new ByteArrayOutputStream()) {

            XWPFTemplate template = XWPFTemplate.compile(is, config)
                    .render(data);
            template.writeAndClose(out);

            byte[] result = out.toByteArray();
            log.info("Word模板填充完成，文件大小: {} bytes", result.length);
            return result;

        } catch (Exception e) {
            log.error("Word模板填充失败", e);
            throw new RuntimeException("Word模板填充失败: " + e.getMessage(), e);
        }
    }
}
```

```java
package com.jy.eleaitender.file.service.impl;

import com.jy.eleaitender.common.dto.response.FileUploadResponse;
import com.jy.eleaitender.common.dto.response.WordStructureVO;
import com.jy.eleaitender.common.entity.file.FileInfo;
import com.jy.eleaitender.common.enums.ResponseCode;
import com.jy.eleaitender.common.exception.FileException;
import com.jy.eleaitender.file.engine.WordStructureParser;
import com.jy.eleaitender.file.engine.WordTemplateEngine;
import com.jy.eleaitender.file.service.IFileStorageService;
import com.jy.eleaitender.file.service.IWordDocumentService;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.io.ByteArrayInputStream;
import java.io.File;
import java.io.FileInputStream;
import java.io.InputStream;
import java.util.Map;

@Slf4j
@Service
public class WordDocumentServiceImpl implements IWordDocumentService {

    private static final String BIZ_TYPE_TENDER_DOC = "tender-document";

    @Autowired
    private IFileStorageService fileStorageService;

    @Autowired
    private WordStructureParser structureParser;

    @Autowired
    private WordTemplateEngine templateEngine;

    @Override
    public WordStructureVO getFileStructure(Long fileId) {
        String filePath = fileStorageService.getFilePath(fileId);
        File file = new File(filePath);
        if (!file.exists()) {
            throw new FileException(ResponseCode.FILE_NOT_FOUND);
        }

        try (InputStream is = new FileInputStream(file)) {
            return structureParser.parse(is);
        } catch (FileException e) {
            throw e;
        } catch (Exception e) {
            log.error("读取文件失败: fileId={}", fileId, e);
            throw new FileException(ResponseCode.FILE_NOT_FOUND);
        }
    }

    @Override
    public Long generateDocument(Long templateFileId, Map<String, Object> data, String fileName) {
        String templatePath = fileStorageService.getFilePath(templateFileId);
        File templateFile = new File(templatePath);
        if (!templateFile.exists()) {
            throw new FileException(ResponseCode.FILE_NOT_FOUND);
        }

        try (InputStream templateStream = new FileInputStream(templateFile)) {
            byte[] result = templateEngine.render(templateStream, data);

            ByteArrayInputStream resultStream = new ByteArrayInputStream(result);
            FileUploadResponse uploadResponse = fileStorageService.uploadFromBytes(
                    resultStream, fileName, BIZ_TYPE_TENDER_DOC);

            log.info("文档生成并上传完成，模板文件ID: {}, 生成文件ID: {}", templateFileId, uploadResponse.getFileId());
            return uploadResponse.getFileId();
        } catch (FileException e) {
            throw e;
        } catch (Exception e) {
            log.error("文档生成失败: templateFileId={}", templateFileId, e);
            throw new RuntimeException("文档生成失败: " + e.getMessage(), e);
        }
    }
}
```

```java
@Override
public FileUploadResponse uploadFromBytes(InputStream inputStream, String fileName, String bizType) {
    try {
        // 生成文件存储路径
        String storagePath = generateStoragePath(fileName, bizType);
        File destFile = new File(storagePath);
        destFile.getParentFile().mkdirs();

        // 写入磁盘
        Files.copy(inputStream, destFile.toPath(), StandardCopyOption.REPLACE_EXISTING);

        // 计算 SHA-256
        String sha256 = DigestUtils.sha256Hex(new FileInputStream(destFile));

        // 检查秒传
        FileInfo existing = getBySha256(sha256);
        if (existing != null) {
            destFile.delete();
            return toUploadResponse(existing);
        }

        // 保存文件记录到数据库
        FileInfo fileInfo = new FileInfo();
        fileInfo.setFileName(fileName);
        fileInfo.setFilePath(storagePath);
        fileInfo.setFileSize(destFile.length());
        fileInfo.setFileSha256(sha256);
        fileInfo.setBizType(bizType);
        // ... 设置 createId/createName/modifyId/modifyName
        fileInfoMapper.insert(fileInfo);

        return toUploadResponse(fileInfo);
    } catch (IOException e) {
        throw new FileException(ResponseCode.FILE_UPLOAD_FAILED);
    }
}
```

```java
        TbProjectTemplate pt = new TbProjectTemplate();

        pt.setTemplateId(supTemplate.getId());
        pt.setTemplateCode(supTemplate.getTemplateCode());
        pt.setTemplateName(supTemplate.getTemplateName());
        pt.setProjectCategory(supTemplate.getProjectCategory());
        pt.setProjectType(supTemplate.getProjectType());
        pt.setFileId(supTemplate.getFileId());
        pt.setContent(supTemplate.getContent());
        pt.setStructureDefinition(supTemplate.getStructureDefinition());
        pt.setVersionNo(supTemplate.getVersionNo());
        projectTemplateMapper.insert(pt);

        // 更新项目 templateId 指向新表
        project.setTemplateId(pt.getId());
        projectMapper.updateById(project);

        log.info("项目绑定模板成功: projectId={}, supTemplateId={}, projectTemplateId={}",
```

脚本新增：

```typescript
const buildTreeData = (chapters: WordChapter[]) => {
  // 将扁平章节列表转为树结构
  const result: any[] = []
  const stack: any[] = []

  for (const ch of chapters) {
    const node = { title: '  '.repeat(ch.level - 1) + ch.title, children: [] }
    while (stack.length >= ch.level) {
      stack.pop()
    }
    if (stack.length > 0) {
      stack[stack.length - 1].children.push(node)
    } else {
      result.push(node)
    }
    stack.push(node)
  }
  return result
}
```
