## Task 1: 修复P0缺陷 - 检测结果JSON解析

**Files:**

- 修改: `ele-ai-tender-system/ele-ai-tender-core/src/main/java/com/jy/eleaitender/core/service/impl/DetectionServiceImpl.java:325-339`

**背景：** AI模块的 `DetectionEngine.toJson()` 将检测结果写入 `ai_detection_record.result`，格式为：
```json
{
  "issues": [
    {
      "position": "第3章第2节",
      "original": "原始文本",
      "suggestion": "修改建议",
      "reason": "原因",
      "severity": "HIGH",
      "detectionType": "SENSITIVE_WORD",
      "policyReference": null,
      "ruleViolated": null
    }
  ],
  "score": 85
}
```

前端 `DetectionReport.vue` 期望的 `DetectionIssueVO` 字段为：
- `recordId`, `detectionType`, `typeName`, `description`, `location`, `suggestion`, `severity`, `handleStatus`

**字段映射关系：**
| AI模块字段 | Core VO字段 | 映射规则 |
|-----------|------------|----------|
| position | location | 直接映射 |
| original + reason | description | 拼接：`original`（如有）+ `reason`（如有） |
| suggestion | suggestion | 直接映射 |
| severity | severity | 直接映射 |
| detectionType | detectionType | 直接映射 |
| (record.id) | recordId | 从record实体取 |
| (DetectionType.getLabel()) | typeName | 枚举转中文 |
| (新增) | handleStatus | 默认0(未处理) |

- [ ] **Step 1: 创建 DetectionResultParser 工具类**

```java
package com.jy.eleaitender.core.util;

import com.fasterxml.jackson.core.type.TypeReference;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.jy.eleaitender.core.dto.response.DetectionIssueVO;
import com.jy.eleaitender.common.enums.DetectionType;
import com.jy.eleaitender.common.entity.core.AiDetectionRecord;
import lombok.extern.slf4j.Slf4j;
import org.springframework.util.StringUtils;

import java.util.*;

@Slf4j
public class DetectionResultParser {

    private static final ObjectMapper MAPPER = new ObjectMapper();

    /**
     * 解析检测结果JSON，提取issue列表
     */
    public static List<DetectionIssueVO> parseIssues(AiDetectionRecord record) {
        if (record == null || !StringUtils.hasText(record.getResult())) {
            return Collections.emptyList();
        }

        try {
            JsonNode root = MAPPER.readTree(record.getResult());
            JsonNode issuesNode = root.get("issues");
            if (issuesNode == null || !issuesNode.isArray()) {
                return Collections.emptyList();
            }

            List<DetectionIssueVO> result = new ArrayList<>();
            for (int i = 0; i < issuesNode.size(); i++) {
                JsonNode issueNode = issuesNode.get(i);
                DetectionIssueVO vo = new DetectionIssueVO();
                vo.setRecordId(record.getId());
                vo.setDetectionType(getStringValue(issueNode, "detectionType", record.getDetectionType()));
                vo.setTypeName(getDetectionTypeName(vo.getDetectionType()));
                vo.setDescription(buildDescription(
                        getStringValue(issueNode, "original", null),
                        getStringValue(issueNode, "reason", null)));
                vo.setLocation(getStringValue(issueNode, "position", null));
                vo.setSuggestion(getStringValue(issueNode, "suggestion", null));
                vo.setSeverity(getStringValue(issueNode, "severity", "MEDIUM"));
                vo.setHandleStatus(0); // 默认未处理
                result.add(vo);
            }
            return result;
        } catch (Exception e) {
            log.error("解析检测结果JSON失败, recordId={}", record.getId(), e);
            return Collections.emptyList();
        }
    }

    /**
     * 解析检测结果JSON，提取score
     */
    public static int parseScore(String resultJson) {
        if (!StringUtils.hasText(resultJson)) {
            return 0;
        }
        try {
            JsonNode root = MAPPER.readTree(resultJson);
            JsonNode scoreNode = root.get("score");
            return scoreNode != null && scoreNode.isInt() ? scoreNode.asInt() : 0;
        } catch (Exception e) {
            return 0;
        }
    }

    /**
     * 解析issue数量
     */
    public static int parseIssueCount(String resultJson) {
        if (!StringUtils.hasText(resultJson)) {
            return 0;
        }
        try {
            JsonNode root = MAPPER.readTree(resultJson);
            JsonNode issuesNode = root.get("issues");
            return issuesNode != null && issuesNode.isArray() ? issuesNode.size() : 0;
        } catch (Exception e) {
            return 0;
        }
    }

    /**
     * 更新result JSON中指定index的issue的handleStatus
     * @return 更新后的JSON字符串，失败返回null
     */
    public static String updateIssueHandleStatus(String resultJson, int issueIndex, int handleStatus) {
        if (!StringUtils.hasText(resultJson)) {
            return null;
        }
        try {
            JsonNode root = MAPPER.readTree(resultJson);
            JsonNode issuesNode = root.get("issues");
            if (issuesNode != null && issuesNode.isArray() && issueIndex >= 0 && issueIndex < issuesNode.size()) {
                ((com.fasterxml.jackson.databind.node.ObjectNode) issuesNode.get(issueIndex))
                        .put("handleStatus", handleStatus);
                return MAPPER.writeValueAsString(root);
            }
            return null;
        } catch (Exception e) {
            log.error("更新检测结果handleStatus失败", e);
            return null;
        }
    }

    /**
     * 批量更新result JSON中所有PENDING状态issue的handleStatus
     * @return 更新后的JSON字符串，失败返回null
     */
    public static String acceptAllIssues(String resultJson) {
        if (!StringUtils.hasText(resultJson)) {
            return null;
        }
        try {
            JsonNode root = MAPPER.readTree(resultJson);
            JsonNode issuesNode = root.get("issues");
            if (issuesNode != null && issuesNode.isArray()) {
                for (int i = 0; i < issuesNode.size(); i++) {
                    JsonNode issue = issuesNode.get(i);
                    JsonNode statusNode = issue.get("handleStatus");
                    if (statusNode == null || statusNode.asInt(-1) == 0) {
                        ((com.fasterxml.jackson.databind.node.ObjectNode) issue).put("handleStatus", 1);
                    }
                }
                return MAPPER.writeValueAsString(root);
            }
            return null;
        } catch (Exception e) {
            log.error("批量接受检测结果失败", e);
            return null;
        }
    }

    private static String getStringValue(JsonNode node, String field, String defaultValue) {
        JsonNode fieldNode = node.get(field);
        return fieldNode != null && !fieldNode.isNull() ? fieldNode.asText() : defaultValue;
    }

    private static String buildDescription(String original, String reason) {
        if (StringUtils.hasText(original) && StringUtils.hasText(reason)) {
            return original + " - " + reason;
        }
        if (StringUtils.hasText(original)) return original;
        if (StringUtils.hasText(reason)) return reason;
        return "";
    }

    private static String getDetectionTypeName(String code) {
        try {
            return DetectionType.fromCode(code).getLabel();
        } catch (Exception e) {
            return code;
        }
    }
}
```

- [ ] **Step 2: 修改 DetectionServiceImpl 的 acceptIssue/rejectIssue/acceptAll 实现**

替换 L200-234：

```java
    @Override
    @Transactional
    public void acceptIssue(Long recordId, Integer issueIndex) {
        AiDetectionRecord record = detectionRecordMapper.selectById(recordId);
        if (record == null) {
            throw new BusinessException(ResponseCode.DETECTION_NOT_FOUND);
        }
        getProjectOrThrow(record.getProjectId());

        String updatedJson = DetectionResultParser.updateIssueHandleStatus(
                record.getResult(), issueIndex, 1); // 1=已接受
        if (updatedJson != null) {
            record.setResult(updatedJson);
            detectionRecordMapper.updateById(record);
        }
        log.info("接受检测建议，记录ID: {}, 问题索引: {}", recordId, issueIndex);
    }

    @Override
    @Transactional
    public void rejectIssue(Long recordId, Integer issueIndex) {
        AiDetectionRecord record = detectionRecordMapper.selectById(recordId);
        if (record == null) {
            throw new BusinessException(ResponseCode.DETECTION_NOT_FOUND);
        }
        getProjectOrThrow(record.getProjectId());

        String updatedJson = DetectionResultParser.updateIssueHandleStatus(
                record.getResult(), issueIndex, 2); // 2=已拒绝
        if (updatedJson != null) {
            record.setResult(updatedJson);
            detectionRecordMapper.updateById(record);
        }
        log.info("拒绝检测建议，记录ID: {}, 问题索引: {}", recordId, issueIndex);
    }

    @Override
    @Transactional
    public void acceptAll(Long projectId) {
        getProjectOrThrow(projectId);
        List<AiDetectionRecord> records = detectionRecordMapper.selectByProjectId(projectId);
        for (AiDetectionRecord record : records) {
            String updatedJson = DetectionResultParser.acceptAllIssues(record.getResult());
            if (updatedJson != null) {
                record.setResult(updatedJson);
                detectionRecordMapper.updateById(record);
            }
            log.info("批量接受检测建议，记录ID: {}", record.getId());
        }
    }
```
