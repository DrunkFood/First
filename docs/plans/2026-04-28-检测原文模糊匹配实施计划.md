# 检测原文模糊匹配 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 解决检测结果中 `original` 原文与 Word 文档实际文本因空格/换行差异导致 >50% 匹配失败的问题

**Architecture:** 在 `WordDocumentFixEngine` 中实现两级匹配策略（精确→规范化），规范化时移除所有空白字符后比较，匹配成功后用位置映射找到原文实际范围再替换。同时优化 AI Prompt 约束原文格式。只改后端匹配逻辑，不改数据结构，完全向后兼容。

**Tech Stack:** Java 21 · Apache POI (XWPFDocument) · Spring Boot 3.2.2

```java
    private byte[] createDocWithText(String text) throws IOException {
        try (XWPFDocument doc = new XWPFDocument();
             ByteArrayOutputStream out = new ByteArrayOutputStream()) {
            XWPFParagraph para = doc.createParagraph();
            XWPFRun run = para.createRun();
            run.setText(text);
            doc.write(out);
            return out.toByteArray();
        }
    }

    private byte[] createDocWithTableText(String header, String value) throws IOException {
        try (XWPFDocument doc = new XWPFDocument();
             ByteArrayOutputStream out = new ByteArrayOutputStream()) {
            var table = doc.createTable(1, 2);
            table.getRow(0).getCell(0).addParagraph().createRun().setText(header);
            table.getRow(0).getCell(1).addParagraph().createRun().setText(value);
            doc.write(out);
            return out.toByteArray();
        }
    }

    private FixReplacement replacement(String original, String targeted) {
        FixReplacement rep = new FixReplacement();
        rep.setOriginal(original);
        rep.setTargeted(targeted);
        return rep;
    }

    private void assertDocContains(byte[] docBytes, String expected) throws IOException {
        try (XWPFDocument doc = new XWPFDocument(new ByteArrayInputStream(docBytes))) {
            boolean found = false;
            for (XWPFParagraph para : doc.getParagraphs()) {
                if (para.getText() != null && para.getText().contains(expected)) {
                    found = true;
                    break;
                }
            }
            if (!found) {
                for (XWPFTable table : doc.getTables()) {
                    for (XWPFTableRow row : table.getRows()) {
                        for (XWPFTableCell cell : row.getTableCells()) {
                            for (XWPFParagraph para : cell.getParagraphs()) {
                                if (para.getText() != null && para.getText().contains(expected)) {
                                    found = true;
                                    break;
                                }
                            }
                            if (found) break;
                        }
                        if (found) break;
                    }
                }
            }
            assertTrue(found, "文档中应包含: " + expected);
        }
    }
}
```

```java
package com.jy.eleaitender.file.engine;

import com.jy.eleaitender.common.dto.FixReplacement;
import lombok.AllArgsConstructor;
import lombok.Data;
import lombok.NoArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.apache.poi.xwpf.usermodel.*;
import org.springframework.stereotype.Component;

import java.io.ByteArrayInputStream;
import java.io.ByteArrayOutputStream;
import java.util.List;

/**
 * Word文档修复引擎
 * 基于Apache POI实现文本查找替换，用于智能检测修复
 */
@Slf4j
@Component
public class WordDocumentFixEngine {

    public FixResult fix(byte[] docxBytes, List<FixReplacement> replacements) {
        int fixedCount = 0;
        int failedCount = 0;

        try (XWPFDocument doc = new XWPFDocument(new ByteArrayInputStream(docxBytes));
             ByteArrayOutputStream out = new ByteArrayOutputStream()) {

            for (FixReplacement rep : replacements) {
                boolean found = replaceInDocument(doc, rep.getOriginal(), rep.getTargeted());
                if (found) {
                    fixedCount++;
                } else {
                    failedCount++;
                    log.warn("未找到原文，跳过修复: original={}",
                            rep.getOriginal().substring(0, Math.min(50, rep.getOriginal().length())));
                }
            }

            doc.write(out);
            return new FixResult(out.toByteArray(), fixedCount, failedCount);
        } catch (Exception e) {
            throw new RuntimeException("Word文档修复失败: " + e.getMessage(), e);
        }
    }

    private boolean replaceInDocument(XWPFDocument doc, String original, String targeted) {
        boolean found = false;

        for (XWPFParagraph para : doc.getParagraphs()) {
            if (replaceInParagraph(para, original, targeted)) {
                found = true;
            }
        }

        for (XWPFTable table : doc.getTables()) {
            for (XWPFTableRow row : table.getRows()) {
                for (XWPFTableCell cell : row.getTableCells()) {
                    for (XWPFParagraph para : cell.getParagraphs()) {
                        if (replaceInParagraph(para, original, targeted)) {
                            found = true;
                        }
                    }
                }
            }
        }

        for (XWPFHeader header : doc.getHeaderList()) {
            for (XWPFParagraph para : header.getParagraphs()) {
                if (replaceInParagraph(para, original, targeted)) {
                    found = true;
                }
            }
        }
        for (XWPFFooter footer : doc.getFooterList()) {
            for (XWPFParagraph para : footer.getParagraphs()) {
                if (replaceInParagraph(para, original, targeted)) {
                    found = true;
                }
            }
        }

        return found;
    }

    /**
     * 替换段落中的文本
     * 两级匹配策略：
     * Level 1: 精确匹配（原有逻辑）
     * Level 2: 规范化匹配（移除所有空白后比较，映射回原文位置替换）
     */
    private boolean replaceInParagraph(XWPFParagraph para, String original, String targeted) {
        String fullText = para.getText();
        if (fullText == null) return false;

        List<XWPFRun> runs = para.getRuns();
        if (runs.isEmpty()) return false;

        // Level 1: 精确匹配
        if (fullText.contains(original)) {
            return doReplace(fullText, original, targeted, runs);
        }

        // Level 2: 规范化匹配
        String actualOriginal = findActualOriginal(fullText, original);
        if (actualOriginal != null) {
            log.debug("规范化匹配成功: original长度={}, 实际原文长度={}",
                    original.length(), actualOriginal.length());
            return doReplace(fullText, actualOriginal, targeted, runs);
        }

        return false;
    }

    private boolean doReplace(String fullText, String actualOriginal,
                              String targeted, List<XWPFRun> runs) {
        XWPFRun firstRun = runs.get(0);
        String replaced = fullText.replace(actualOriginal, targeted);

        for (XWPFRun run : runs) {
            run.setText("", 0);
        }
        firstRun.setText(replaced, 0);
        return true;
    }

    /**
     * 在原始文本中找到与 original 规范化等价的子串
     * 原理：对双方都移除所有空白后比较，匹配成功后通过位置映射还原原始文本中的实际范围
     *
     * @param rawText  段落原始文本
     * @param original AI返回的原文
     * @return 原始文本中对应的子串，未找到返回 null
     */
    static String findActualOriginal(String rawText, String original) {
        String normalizedRaw = normalize(rawText);
        String normalizedOriginal = normalize(original);

        int normStart = normalizedRaw.indexOf(normalizedOriginal);
        if (normStart < 0) return null;
        int normEnd = normStart + normalizedOriginal.length();

        // 构建规范化位置 → 原始位置的映射
        // normToRaw[i] = 规范化字符i在原始文本中的起始位置
        int[] normToRaw = new int[normalizedRaw.length() + 1];

        int rawIdx = 0;
        int normIdx = 0;

        while (normIdx <= normalizedRaw.length() && rawIdx <= rawText.length()) {
            // 跳过原始文本中的空白（规范化时被移除）
            if (rawIdx < rawText.length() && isWhitespaceChar(rawText.charAt(rawIdx))) {
                rawIdx++;
                continue;
            }

            normToRaw[normIdx] = rawIdx;

            if (normIdx == normalizedRaw.length()) break;

            // 非空白字符：一一对应
            rawIdx++;
            normIdx++;
        }

        return rawText.substring(normToRaw[normStart], normToRaw[normEnd]);
    }

    /**
     * 规范化文本：移除所有空白字符
     * 中文文本中空格/换行通常是格式差异，移除后比较可以消除这些差异
     */
    static String normalize(String text) {
        if (text == null) return "";
        return text.replaceAll("\\s+", "");
    }

    private static boolean isWhitespaceChar(char c) {
        return Character.isWhitespace(c) || c == ' ' || c == '　';
    }

    @Data
    @AllArgsConstructor
    @NoArgsConstructor
    public static class FixResult {
        private byte[] documentBytes;
        private int fixedCount;
        private int failedCount;
    }
}
```
